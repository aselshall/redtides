import pandas as pd

# Define bloom threshold and gap tolerance
BLOOM_THRESHOLD = 1
MAX_GAP = pd.Timedelta(days=14)

# Initialize bloom state
bloom_id = 0
in_bloom = False
last_bloom_date = None

bloom_list = []

for index, row in df.iterrows():
    cellcount = row['CELLCOUNT_1e5']
    date = row.name
    
    if pd.notna(cellcount) and cellcount >= BLOOM_THRESHOLD:
        if not in_bloom or (last_bloom_date and (date - last_bloom_date) > MAX_GAP):
            # Start a new bloom only if there's a large gap
            bloom_id += 1
            in_bloom = True
        last_bloom_date = date
    
    elif last_bloom_date and (date - last_bloom_date) <= MAX_GAP:
        # Continue the bloom if within max gap
        in_bloom = True
    
    else:
        in_bloom = False
    
    bloom_list.append(bloom_id if in_bloom else 0)

# ✅ Add bloom column to dataframe
df['BLOOM'] = bloom_list

# ✅ Fix short gaps within an active bloom using forward fill
# This will extend a bloom across short gaps or low values
df['BLOOM'] = df['BLOOM'].replace(to_replace=0, method='ffill', limit=2)

# ✅ Resample after processing (weekly)
df_weekly = df.resample('W').agg({
    'CELLCOUNT_1e5': 'max',   # Take the max weekly cell count
    'BLOOM': 'first'          # Take the first bloom ID of the week
})

# ✅ Ensure bloom IDs are consistent after resampling
df_weekly['BLOOM'] = df_weekly['BLOOM'].where(df_weekly['BLOOM'] > 0)
df_weekly['BLOOM'] = df_weekly['BLOOM'].ffill()

# ✅ Ensure bloom IDs remain consistent across short gaps
df_weekly['BLOOM'] = df_weekly['BLOOM'].where(
    (df_weekly['BLOOM'] != df_weekly['BLOOM'].shift()) |
    (df_weekly['CELLCOUNT_1e5'] >= BLOOM_THRESHOLD)
)

# ✅ Save to CSV or continue with further processing
df_weekly.to_csv('weekly_summary.csv')
